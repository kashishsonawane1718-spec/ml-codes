# Practical 3: Multiple Linear Regression
# Dataset: California Housing Dataset
# ==========================================================

# Step 1: Import Libraries

import pandas as pd
import numpy as np

from sklearn.datasets import fetch_california_housing
from sklearn.model_selection import train_test_split
from sklearn.linear_model import LinearRegression
from sklearn.metrics import mean_squared_error, r2_score

# Step 2: Load Dataset

housing = fetch_california_housing()

# Separate Input and Output

X = housing.data
y = housing.target

# Create DataFrame

df = pd.DataFrame(X, columns=housing.feature_names)
df["HousePrice"] = y

print("First Five Records:")
print(df.head())

print("\nDataset Shape:", df.shape)

# Step 3: Split Dataset into Training and Testing Data

X_train, X_test, y_train, y_test = train_test_split(
    X,
    y,
    test_size=0.20,
    random_state=42
)

print("\nTraining Data Shape:", X_train.shape)
print("Testing Data Shape:", X_test.shape)

# Step 4: Create Multiple Linear Regression Model

model = LinearRegression()

# Step 5: Train the Model

model.fit(X_train, y_train)

print("\nModel Training Completed Successfully!")

# Step 6: Predict House Prices

y_pred = model.predict(X_test)

# Step 7: Evaluate the Model

mse = mean_squared_error(y_test, y_pred)

rmse = np.sqrt(mse)

r2 = r2_score(y_test, y_pred)

# Step 8: Display Model Performance

print("\n========== Model Performance ==========")

print("Mean Squared Error (MSE):", round(mse, 4))

print("Root Mean Squared Error (RMSE):", round(rmse, 4))

print("R2 Score:", round(r2, 4))

# Step 9: Display Intercept

print("\nIntercept:", round(model.intercept_, 4))

# Step 10: Display Regression Coefficients

print("\n========== Regression Coefficients ==========")

for feature, coef in zip(housing.feature_names, model.coef_):
    print(feature, ":", round(coef, 4))

# Step 11: Compare Actual and Predicted Values

results = pd.DataFrame({
    "Actual Price": y_test[:10],
    "Predicted Price": y_pred[:10]
})

print("\n========== Actual vs Predicted Prices ==========")
print(results)
